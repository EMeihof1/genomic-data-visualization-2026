clusters_to_test <- c(2, 4, 6)

pvals <- lapply(clusters_to_test, function(cl) {
  cells_in  <- names(clusters)[clusters == cl]
  cells_out <- names(clusters)[clusters != cl]
  
  data.frame(
    protein = colnames(mat),
    pvalue  = sapply(colnames(mat), function(p) {
      wilcox.test(mat[cells_in, p],
                  mat[cells_out, p],
                  alternative = "greater")$p.value
    }),
    cluster = paste0("Cluster ", cl)
  )
})
